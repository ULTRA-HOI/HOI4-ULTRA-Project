# convert 1940 to 1942 tank template
country_event = {
	id = ger_templates.21
	hidden = yes

	is_triggered_only = yes

	trigger = {
		tag = GER
		is_ai = yes
		check_variable = { 1942_tanks_to_convert > 0 }
	}

	option = {
		set_temp_variable = { div_count = 5 }

		set_temp_variable = { ai_overconverting = num_battalions_with_type@heavy_armor }
