# Final Operation Valkyrie Event
country_event = {
	id = valkyrie.100
	title = valkyrie.100.title
	desc = valkyrie.100.desc
	picture = GFX_report_event_GER_black_reichswehr_forces
	
	is_triggered_only = yes
	
	option = {
		name = valkyrie.100.option.a # "The Führer survives!"
		if = {
				limit = {
					has_country_leader_with_trait = GER_der_fuhrer_fuhrerprinzip_wehrmacht
				}
				random_character = {
					limit = {
						is_country_leader = yes
					}
					swap_country_leader_traits = {
						remove = GER_der_fuhrer_fuhrerprinzip_wehrmacht
						add = GER_der_fuhrer_late_ultra
					}
					set_portraits = {
						civilian = { large = GFX_portrait_GER_adolf_hitler_late }
					}
				}
			}
		# Remove involved generals
		retire_character = GER_erich_hoepner
		retire_character = GER_erwin_von_witzleben
		retire_character = GER_alexander_von_falkenhausen
		retire_character = GER_hans_karl_freiherr_von_esebeck
		retire_character = GER_erwin_rommel
		
		# Set Balance of Power
		if = {
			limit = { 
				is_power_balance_side_active = {
					id = wehrmacht_waffen_ss_balance
					side = waffen_ss_side
				}
			}
			GER_bop_absurdly_high_increase_effect = yes
		}
		else_if = {
			limit = {
				is_power_balance_side_active = {
						id = wehrmacht_waffen_ss_balance
						side = wehrmacht_side
					}
			}
			remove_all_power_balance_modifiers = {
				id = wehrmacht_waffen_ss_balance
			}
		}
	}
	option = {
		name = valkyrie.100.option.b # "The Führer is dead! Göring takes over!"
		retire_country_leader = yes
		promote_character = { 
			character = GER_hermann_goring
			ideology = nazism
		}
		add_stability = -0.1
		add_war_support = -0.1
		# Remove involved generals
		retire_character = GER_erich_hoepner
		retire_character = GER_erwin_von_witzleben
		retire_character = GER_alexander_von_falkenhausen
		retire_character = GER_hans_karl_freiherr_von_esebeck
		retire_character = GER_erwin_rommel
		# Set Balance of Power
		if = {
			limit = { 
				is_power_balance_side_active = {
					id = wehrmacht_waffen_ss_balance
					side = waffen_ss_side
				}
			}
			GER_bop_absurdly_high_increase_effect = yes
		}
		else_if = {
			limit = {
				is_power_balance_side_active = {
						id = wehrmacht_waffen_ss_balance
						side = wehrmacht_side
					}
			}
			GER_bop_absurdly_high_decrease_effect = yes
		}
		hidden_effect = { 
			set_cosmetic_tag = GER_nazi_gvt
		}	
	}
	
	option = {
		name = valkyrie.100.option.c # "Hitler is dead, Valkyrie succeeds!"
		promote_character = { 
			character = GER_ludwig_beck
			ideology = centrism
		}
		add_popularity = {
			ideology = neutrality
			popularity = 0.55
		}
		retire_country_leader = yes

		set_politics = {
			ruling_party = neutrality
			last_election = "1933.3.5"
			election_frequency = 48
			elections_allowed = no
		}
		add_stability = -0.2
		add_war_support = -0.1
		country_event = {
			id = valkyrie.101
			days = 7
		}

		# Remove all SS generals
		every_unit_leader = {
			limit = {
				has_trait = GER_ss
			}
			remove_unit_leader = yes
		}

		# Remove all pro Nazi MHC
		retire_character = GER_wilhelm_keitel
		retire_character = GER_walther_von_brauchitsch
		retire_character = GER_franz_halder

		# Set Balance of Power
		if = {
			limit = { 
				is_power_balance_in_range = {
					id = wehrmacht_waffen_ss_balance
					range = waffen_ss_dominant
				}
			}
			set_country_flag = GER_nazi_resistance_very_active_flag
			remove_all_power_balance_modifiers = {
				id = wehrmacht_waffen_ss_balance
			}
		}
		else_if = {
			limit = { 
				is_power_balance_in_range = {
					id = wehrmacht_waffen_ss_balance
					range = waffen_ss_strong
				}
			}
			set_country_flag = GER_nazi_resistance_active_flag
			remove_all_power_balance_modifiers = {
				id = wehrmacht_waffen_ss_balance
			}
		}
		else_if = {
			limit = { 
				is_power_balance_in_range = {
					id = wehrmacht_waffen_ss_balance
					range = waffen_ss_favored
				}
			}
			set_country_flag = GER_nazi_resistance_somewhat_active_flag
			remove_all_power_balance_modifiers = {
				id = wehrmacht_waffen_ss_balance
			}
		}	
		else_if = {
			limit = {
				is_power_balance_side_active = {
						id = wehrmacht_waffen_ss_balance
						side = wehrmacht_side
					}
			}
			GER_bop_absurdly_high_decrease_effect = yes
			set_power_balance_gfx = {
				id = wehrmacht_waffen_ss_balance
				side = wehrmacht_side
				gfx = GFX_ger_bop_wh_max
			}
		}
		hidden_effect = { 
			set_cosmetic_tag = GER_military_gvt
			set_party_name = {
				ideology = neutrality
				name = GER_military_gvt_neutrality_party
				long_name = GER_military_gvt_neutrality_party_long
			}
			set_party_name = {
				ideology = fascism
				name = GER_military_gvt_fascism_party
				long_name = GER_military_gvt_fascism_party_long
			}
		}
	}
}
