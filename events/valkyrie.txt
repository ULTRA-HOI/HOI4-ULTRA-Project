# 13th March 1943 - Smolensk airport attempt (Major)
country_event = {
	id = valkyrie.8
	title = valkyrie.8.title
	desc = valkyrie.8.desc
	picture = GFX_report_event_generic_military_ceremony
	
	is_triggered_only = yes
	
	trigger = {
		has_country_leader = { name = "Adolf Hitler" }
		SOV = { exists = yes }
		OR = {
			AND = { 
			controls_state = 242  # Smolensk
			controls_state = 207  # Vitebsk
			controls_state = 241  # Gomel
			controls_state = 1036 # Riga
			controls_state = 1040 # Tallinn
			}
			AND = {
				NOT = {
					controls_state = 11  # Kaunas
					controls_state = 202 # Kyiv
				}
			}	
		}
	}
	
	option = {
		name = valkyrie.8.option.a # "There are too many guards."
		country_event = {
			id = valkyrie.9
			days = 1
		}
	}
	
	option = {
		name = valkyrie.8.option.b # "Open fire!"
		complete_national_focus = GER_operation_valkyrie
	}
}

# Final Operation Valkyrie Event
country_event = {
	id = valkyrie.100
	title = valkyrie.100.title
	desc = valkyrie.100.desc
	picture = GFX_report_event_generic_bomb_2
	
	is_triggered_only = yes
	
	option = {
		name = valkyrie.100.option.a # "The Führer survives!"
		if = {
				limit = {
					has_country_leader_with_trait = GER_der_fuhrer_fuhrerprinzip_wehrmacht
				}
				random_character = {
					limit = {
						is_country_leader = yes
					}
					swap_country_leader_traits = {
						remove = GER_der_fuhrer_fuhrerprinzip_wehrmacht
						add = GER_der_fuhrer_late_ultra
					}
					set_portraits = {
						civilian = { large = GFX_portrait_GER_adolf_hitler_late }
					}
				}
			}
		# Remove involved generals
		retire_character = GER_erich_hoepner
		retire_character = GER_erwin_von_witzleben
		retire_character = GER_alexander_von_falkenhausen
		retire_character = GER_hans_karl_freiherr_von_esebeck
		retire_character = GER_erwin_rommel
		
		# Remove Balance of Power
		 remove_all_power_balance_modifiers = {
			id = wehrmacht_waffen_ss_balance
		}
	}
	
	option = {
		name = valkyrie.100.option.b # "The Führer is dead! Göring takes over!"
		retire_country_leader = yes
		promote_character = { 
			character = GER_hermann_goring
			ideology = nazism
		}
		add_stability = -0.1
		add_war_support = -0.1
		# Remove involved generals
		retire_character = GER_erich_hoepner
		retire_character = GER_erwin_von_witzleben
		retire_character = GER_alexander_von_falkenhausen
		retire_character = GER_hans_karl_freiherr_von_esebeck
		retire_character = GER_erwin_rommel
		# Remove Balance of Power
		 remove_all_power_balance_modifiers = {
			id = wehrmacht_waffen_ss_balance
		}
	}
	
	option = {
		name = valkyrie.100.option.c # "Hitler is dead, Valkyrie succeeds!"
		promote_character = { 
			character = GER_ludwig_beck
			ideology = centrism
		}
		add_popularity = {
			ideology = neutrality
			popularity = 0.55
		}
		retire_country_leader = yes

		set_politics = {
			ruling_party = neutrality
			last_election = "1933.3.5"
			election_frequency = 48
			elections_allowed = no
		}
		add_stability = -0.2
		add_war_support = -0.1
		country_event = {
			id = valkyrie.101
			days = 7
		}

		# Remove all SS generals
		every_unit_leader = {
			limit = {
				has_trait = GER_ss
			}
			remove_unit_leader = yes
		}
		# Remove Balance of Power
		 remove_all_power_balance_modifiers = {
			id = wehrmacht_waffen_ss_balance
		}
	}
}
