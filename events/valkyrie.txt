# Final Operation Valkyrie Event
country_event = {
	id = valkyrie.100
	title = valkyrie.100.title
	desc = valkyrie.100.desc
	picture = GFX_report_event_generic_bomb_2
	
	is_triggered_only = yes
	
	option = {
		name = valkyrie.100.option.a # "The Führer survives!"
		# Remove involved generals
		remove_unit_leader = 123 # Stauffenberg
		remove_unit_leader = 124 # Tresckow
		remove_unit_leader = 125 # Beck
		remove_unit_leader = 126 # Rommel
		# Remove Balance of Power
		remove_power_balance = {
			id = wehrmacht_waffen_ss_balance
		}
	}
	
	option = {
		name = valkyrie.100.option.b # "The Führer is dead! Himmler takes over!"
		retire_country_leader = yes
		create_country_leader = {
			name = "Heinrich Himmler"
			desc = "POLITICS_HIMMLER_DESC"
			picture = "Portrait_Germany_Heinrich_Himmler.dds"
			expire = "1965.1.1"
			ideology = nazism
			traits = {
				# Himmler traits
			}
		}
		# Remove involved generals
		remove_unit_leader = 123 # Stauffenberg
		remove_unit_leader = 124 # Tresckow
		remove_unit_leader = 125 # Beck
		remove_unit_leader = 126 # Rommel
		# Remove Balance of Power
		remove_power_balance = {
			id = wehrmacht_waffen_ss_balance
		}
	}
	
	option = {
		name = valkyrie.100.option.c # "Hitler is dead, Valkyrie succeeds!"
		retire_country_leader = yes
		create_country_leader = {
			name = "Ludwig Beck"
			desc = "POLITICS_BECK_DESC"
			picture = "Portrait_Germany_Ludwig_Beck.dds"
			expire = "1965.1.1"
			ideology = conservatism
			traits = {
				# Beck traits
			}
		}
		# Remove all SS generals
		every_unit_leader = {
			limit = {
				has_trait = GER_ss
			}
			remove_unit_leader = THIS
		}
		# Remove Balance of Power
		remove_power_balance = {
			id = wehrmacht_waffen_ss_balance
		}
	}
}
