ability = {
	light_field_gun_tier_1 = {
		name = ABILITY_LIGHT_FIELD_GUN
		desc = ABILITY_LIGHT_FIELD_GUN_DESC
		
		sound_effect = command_power_ability_offensive
		
		type = army_leader
		
		allowed = {
			FROM = {
				NOT = {
					has_country_flag = light_field_gun
				}
				has_equipment = {
					light_field_gun_equipment > 5
				}
			}

			#Overstaffed Generals loose access to Field Guns
			set_temp_variable = { max_unit_limit = 11 } #Needs to be actual max + 1
			#Modifiing the max from various different traits
			if = { limit = { has_trait = panzer_expert } subtract_from_temp_variable = { var = max_unit_limit value = 2 } }
			if = { limit = { has_trait = combined_arms_expert } subtract_from_temp_variable = { var = max_unit_limit value = 3 } }
			if = { limit = { has_trait = invader_ii } subtract_from_temp_variable = { var = max_unit_limit value = 1 } }
			if = { limit = { has_trait = skirmisher } subtract_from_temp_variable = { var = max_unit_limit value = 1 } }
			if = { limit = { has_trait = paratrooper } subtract_from_temp_variable = { var = max_unit_limit value = 3 } }
			if = { limit = { has_trait = camouflage_expert } subtract_from_temp_variable = { var = max_unit_limit value = 1 } }
			if = { limit = { has_trait = skilled_staffer } add_to_temp_variable = { var = max_unit_limit value = 2 } }
			if = { limit = { has_trait = skilled_staffer_sov } add_to_temp_variable = { var = max_unit_limit value = 4 } }
			if = { limit = { has_trait = SOV_hq } add_to_temp_variable = { var = max_unit_limit value = 4 } }
			num_units < max_unit_limit

			is_leading_army_group = no
			is_border_war = no
		}
		
		unit_modifiers = {
			org_loss_when_moving = 0.1
		}
		ai_will_do = {
			factor = -1
			modifier = {
				FROM = {
					command_power > 1.5
				}
				check_variable = { num_units_in_combat > 6 }
				
				set_temp_variable = { temp = avg_combat_status }
				check_variable = { temp > 0.10 }
				check_variable = { ai_random > temp }
				
				add = 2
			}
		}
		
		cost = 0.03
		duration = 72
		cooldown = 120
		
		one_time_effect = {
			IF = {
				LIMIT = {
					FROM = {
						is_ai = yes
					}
				}
				FROM = {
					add_equipment_to_stockpile = {
    					type = light_field_gun_equipment
    					amount = -3
					}
				}
			}
			ELSE = {
				FROM = {
					add_equipment_to_stockpile = {
    					type = light_field_gun_equipment
    					amount = -6
					}
				}
			}
			add_temporary_buff_to_units = {
				combat_offense = 0.02
				combat_breakthrough = 0.10
				combat_defense = 0.02
				combat_entrenchment = 0.10

				days = 3
				tooltip = ABILITY_FIELD_GUN_TOOLTIP
			}
		}
	}
