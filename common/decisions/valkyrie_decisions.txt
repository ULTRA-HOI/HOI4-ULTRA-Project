valkyrie_subversion_decisions = {

	valkyrie_subversion_decision = {
		icon = decision_generic_conspiracy
		
		available = {
			has_war_with = SOV
			has_completed_focus = GER_subverting_valkyrie
		}
		
		visible = {
			has_completed_focus = GER_subverting_valkyrie
		}
		
		fire_only_once = no
		
		days_remove = 90
		days_re_enable = 1
		
		modifier = {
			# Visual indicator that subversion is active
			political_power_gain = -0.1
		}
		
		remove_effect = {
			if = {
				limit = {
					has_war_with = SOV
				}
				# Shift Balance of Power towards Wehrmacht
				add_power_balance_value = {
					id = wehrmacht_waffen_ss_balance
					value = -0.05
				}
				# Restart the decision
				activate_decision = valkyrie_subversion_decision
			}
		}
		
		ai_will_do = {
			factor = 100
		}
	}
}
