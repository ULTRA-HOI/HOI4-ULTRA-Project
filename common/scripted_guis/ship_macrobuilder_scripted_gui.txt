	ship_macrobuilder_button_scripted_gui = {
		context_type = player_context
		
		window_name = "ship_macrobuilder_button_containter"
		parent_window_name = production_win_top

		
		properties = {
			macrobuilder_button = {
				frame = ship_macrobuilder_button_frame
			}
		}

		triggers = {
			macrobuilder_button_visible = {
				if = {
					limit = { has_country_flag = ship_macrobuilder_visible }
					set_temp_variable = { ship_macrobuilder_button_frame = 2 }
				}
				else = {
					set_temp_variable = { ship_macrobuilder_button_frame = 1 }
				}
			}
		}

		effects = {
			macrobuilder_button_click = {
				if = {
					limit = { has_country_flag = ship_macrobuilder_visible }
					clr_country_flag = ship_macrobuilder_visible
				}
				else = {
					set_country_flag = ship_macrobuilder_visible
				}
				init_ship_macrobuilder = yes
			}
		}
	}	
	
	ship_macrobuilder_scripted_gui = {
		context_type = player_context
		
		window_name = "ship_macrobuilder_containter"
		parent_window_name = navy_repair_window

		visible = {
			has_country_flag = ship_macrobuilder_visible
		}

		dynamic_lists = {
			macrobuilder_ships = {
				array = ship_macrobuilder_entries

				entry_container = "macrobuilder_ship_entry"

		triggers = {
			equipment_button_click_enabled = {
				meta_trigger = {
					text = {
						has_tech = [TECH]

				}
			}
			build_x5_button_click_enabled = {
				meta_trigger = {
					text = {
						has_tech = [TECH]
					}
					TECH = "[?ship_macrobuilder_techs^i.GetTokenKey]"
				}
			}
			build_x10_button_click_enabled = {
				meta_trigger = {
					text = {
						has_tech = [TECH]
					}
					TECH = "[?ship_macrobuilder_techs^i.GetTokenKey]"
				}
			}
		}

		effects = {
			equipment_button_click = {
				macro_build_ship = yes
			}
			build_x5_button_click = {
				macro_build_ship = yes
				macro_build_ship = yes
				macro_build_ship = yes
				macro_build_ship = yes

				macro_build_ship = yes
				macro_build_ship = yes

				macro_build_ship = yes
				macro_build_ship = yes
				macro_build_ship = yes
			}
			assign_mio_button_click = {
				set_variable = {
					ship_macrobuilder_selected_ship_index = i
				}
			}
			assign_mio_button_right_click = {
				log = "Removing MIO, Ship Index: [?i]"
				set_variable = { ROOT.ship_macrobuilder_assigned_mios^i = 0 }
			}
			close_button_click = {
				clr_country_flag = ship_macrobuilder_visible
			}
		}

		properties = {
			ship_icon = {

			}

			assign_mio_button = {
				image = [ship_macrobuilder_assigned_mio_icon]
			}
		}
	}
	
	ship_macrobuilder_mio_scripted_gui = {
		context_type = player_context
		
		window_name = "ship_macrobuilder_mio_containter"
		parent_window_name = ship_macrobuilder_containter_instance

		visible = {
			has_variable = ship_macrobuilder_selected_ship_index
		}

		dynamic_lists = {
			macrobuilder_mios = {
				array = ship_macrobuilder_mios

				change_scope = no

				entry_container = "macrobuilder_mio_entry"
			}
		}

		triggers = {
			mio_button_click_enabled = {
				meta_trigger = {
					text = {
						mio:[MIO] = {
							set_temp_variable = { PREV.used = number_of_currently_assigned_tasks }
							set_temp_variable = { PREV.total = max_task_capacity }
						}
					}
					MIO = "[?v.GetTokenKey]"
				}
				check_variable = { used < total }
			}
		}

		effects = {
			mio_button_click = {
				log = "Setting MIO: [?v.GetTokenKey], Ship Index: [?ship_macrobuilder_selected_ship_index]"
				set_variable = {
					ROOT.ship_macrobuilder_assigned_mios^ROOT.ship_macrobuilder_selected_ship_index = v
				}
				clear_variable = ship_macrobuilder_selected_ship_index
			}
			close_button_click = {
				clear_variable = ship_macrobuilder_selected_ship_index
			}
		}

		properties = {
			mio_icon = {
				image = [ship_macrobuilder_mio_icon]
			}
		}
	}
