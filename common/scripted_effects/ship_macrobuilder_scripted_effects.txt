macro_build_ship = {
	meta_effect = {
		text = {
			set_variable = { type_var = token:[CLASS_NAME] }
		}
		# debug = yes
		CLASS_NAME = "[ship_macrobuilder_class_name]"
	}
	log = "[?type_var]"
	log = "[?type_var.GetTokenKey]"
	log = "[?type_var.GetTokenLocalizedKey]"
	meta_effect = {
		text = {
			add_equipment_production = {
				equipment = {
					type = [TYPE]
					creator = [TAG]
					version_name = "[CLASS]"
				}
				[MIO]
				requested_factories = [DOCKYARDS]
				progress = 0.0
				amount = 1
			}
		}
		debug = yes
		CLASS = "[?type_var.GetTokenLocalizedKey]"
		TYPE = "[?v.GetTokenKey]"
		VAR = "[?v.GetTokenKey]"
		TAG = "[ROOT.GetTag]"
		MIO = "[ship_macrobuilder_selected_mio]"
		DOCKYARDS = "[ship_macrobuilder_assigned_dockyards]"
	}
	clear_variable = type_var
}
