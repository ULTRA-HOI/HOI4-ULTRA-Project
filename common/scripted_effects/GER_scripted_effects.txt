		id = wehrmacht_waffen_ss_balance
		left_side = wehrmacht_side # will keep the last set side if not specified
		right_side = waffen_ss_side # will keep the last set side if not specified
		set_default = yes # whether or not to reset the power balance to the initial value, default = no

 }
