logistical_calculations = {
	if = {
		limit = { NOT = { has_tech = motorised_infantry } }
		set_technology  = { motorised_infantry = 1 }
	}
	
	light_logistics_calculations = yes
	heavy_logistics_calculations = yes
	mot_logistics_calculations = yes
	support_logistics_calculations = yes

	set_variable = { total_trucks = total_light_trucks }
	add_to_variable = { total_trucks = total_heavy_trucks }
	add_to_variable = { total_trucks = total_mot_trucks }
	add_to_variable = { total_trucks = total_support_trucks }
	
	set_temp_variable = { truck_diff = total_trucks }
	subtract_from_temp_variable = { truck_diff = logi_trucks }
	clamp_temp_variable = {
		var = truck_diff
		max = num_equipment@motorized_equipment
	}
	multiply_temp_variable = { truck_diff = -1 }
	add_equipment_to_stockpile = {
		type = motorized_equipment
		amount = truck_diff
	}
	subtract_from_variable = { logi_trucks = truck_diff }
	
	#Applies equipment needs
	remove_motorisation_techs = yes
	calc_motorisation_techs = yes
	set_motorisation_techs = yes
}

add_starting_logistics_trucks = {
	set_variable = { logi_trucks = total_trucks }
	set_variable = { truck_attrition_30_days = 0 }
	add_to_array = { attrition_array = 0 }
	resize_array = { attrition_array = 30 }
}

clear_motorisation_techs = {
	for_loop_effect = {
		start = 0
		end = 50
		compare = less_than_or_equals
		value = temp
		meta_effect = {
			text = {
				set_technology  = { light_log_tech_[LEVEL] = 0 popup = no }
				set_technology  = { heavy_log_tech_[LEVEL] = 0 popup = no }
				set_technology  = { mot_log_tech_[LEVEL] = 0 popup = no }
				set_technology  = { sup_log_tech_[LEVEL] = 0 popup = no }
			}
			LEVEL = "[?temp]"
		}
	}
}

calc_motorisation_techs = {
	set_variable = { supply_mult = logi_trucks }
	divide_variable = { supply_mult = total_trucks }
	set_temp_variable = { supply_mult_scale_factor = 0.1 }
	subtract_from_variable = { supply_mult = supply_mult_scale_factor }
	multiply_temp_variable = { supply_mult_scale_factor = -1 }
	add_to_temp_variable = { supply_mult_scale_factor = 1 }
	divide_variable = { supply_mult = supply_mult_scale_factor }
	clamp_variable = {
		var = supply_mult
		min = 0
		max = 1
	}
	
	set_temp_variable = { tech_granularity_mult = 10 }
	
	set_variable = { light_logistics_effective_motorisation_level = light_logistics_motorisation_level }
	multiply_variable = { light_logistics_effective_motorisation_level = tech_granularity_mult }
	multiply_variable = { light_logistics_effective_motorisation_level = supply_mult }
	round_variable = light_logistics_effective_motorisation_level
	
	set_variable = { heavy_logistics_effective_motorisation_level = heavy_logistics_motorisation_level }
	multiply_variable = { heavy_logistics_effective_motorisation_level = tech_granularity_mult }
	multiply_variable = { heavy_logistics_effective_motorisation_level = supply_mult }
	round_variable = heavy_logistics_effective_motorisation_level
	
	set_variable = { mot_logistics_effective_motorisation_level = mot_logistics_motorisation_level }
	multiply_variable = { mot_logistics_effective_motorisation_level = tech_granularity_mult }
	multiply_variable = { mot_logistics_effective_motorisation_level = supply_mult }
	round_variable = mot_logistics_effective_motorisation_level
	
	set_variable = { sup_logistics_effective_motorisation_level = sup_logistics_motorisation_level }
	multiply_variable = { sup_logistics_effective_motorisation_level = tech_granularity_mult }
	multiply_variable = { sup_logistics_effective_motorisation_level = supply_mult }
	round_variable = sup_logistics_effective_motorisation_level
}

logi_trucks_attrition_calculations = {
	if = {
		limit = { THIS = { has_war = yes } }
		set_temp_variable = { attrition_today = logi_trucks }
		set_temp_variable = { logi_truck_daily_attrition_scale_divisor = 1000 }
		divide_temp_variable = { attrition_today = logi_truck_daily_attrition_scale_divisor }
		add_to_temp_variable = { attrition_today = attrition_rollover }
		set_variable = { attrition_rollover = attrition_today }
		subtract_from_temp_variable = { attrition_today = 0.5 }
		round_temp_variable = attrition_today
		subtract_from_variable = { attrition_rollover = attrition_today }
	}
	else = {
		set_temp_variable = { attrition_today = 0 }
	}
	remove_from_array = {
		array = attrition_array
		index = 0
	}
	add_to_array = {
		array = attrition_array
		value = attrition_today
	}
	add_to_variable = { truck_attrition_30_days = attrition_today }
	subtract_from_variable = { logi_trucks = attrition_today }
}
