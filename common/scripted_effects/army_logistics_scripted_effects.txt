calc_motorisation_techs = {
	set_temp_variable = { tech_granularity_mult = 10 }
	for_loop_effect = {
		end = num_logi_categories
		value = logi_category

		meta_effect = {
			text = {
				set_variable = { [LOGI_CAT]_supply_mult = [LOGI_CAT]_logi_trucks }
				divide_variable = { [LOGI_CAT]_supply_mult = total_[LOGI_CAT]_trucks }
				set_temp_variable = { supply_mult_scale_factor = 0.1 }
				subtract_from_variable = { [LOGI_CAT]_supply_mult = supply_mult_scale_factor }
				multiply_temp_variable = { supply_mult_scale_factor = -1 }
				add_to_temp_variable = { supply_mult_scale_factor = 1 }
				divide_variable = { [LOGI_CAT]_supply_mult = supply_mult_scale_factor }
				clamp_variable = {
					var = [LOGI_CAT]_supply_mult
					min = 0
					max = 1
				}
				if = {
					limit = { check_variable = { total_[LOGI_CAT_trucks] = 0 } }
					set_variable = { [LOGI_CAT]_supply_mult = 1 }
				}
				
				set_variable = { [LOGI_CAT]_logistics_effective_motorisation_level = [LOGI_CAT]_logistics_motorisation_level }
				multiply_variable = { [LOGI_CAT]_logistics_effective_motorisation_level = tech_granularity_mult }
				multiply_variable = { [LOGI_CAT]_logistics_effective_motorisation_level = [LOGI_CAT]_supply_mult }
				round_variable = [LOGI_CAT]_logistics_effective_motorisation_level

			}
			LOGI_CAT = "[get_logistics_category]"
		}
	}
}

					subtract_from_variable = { [LOGI_CAT]_attrition_rollover = attrition_today }
				}
				else = {
					set_temp_variable = { attrition_today = 0 }
				}

update_logi_truck_stockpiles = {
	excess_trucks_to_stockpile = yes
	set_temp_variable = { trucks_remaining = num_equipment@motorized_equipment }
	# make sure there isn't a false surplus of trucks due to e.g. division disbandment when actually in a deficit
	add_to_temp_variable = { trucks_remaining = num_equipment_in_armies@motorized_equipment }
	subtract_from_temp_variable = { trucks_remaining = num_target_equipment_in_armies@motorized_equipment }

	set_temp_variable = { trucks_transferred = trucks_remaining }

	if = {
		limit = {
			check_variable = { trucks_remaining > 0 }
		}
		# loop through each truck reinforcement priority, starting with high priority
		for_loop_effect = {
			start = 2
			end = 0
			compare = greater_than_or_equals
			add = -1
			value = priority
			break = break_loop

			# loop through each motorisation category
			for_loop_effect = {
				end = num_logi_categories
				value = logi_category
							
				# only want to consider the same priority categories as whats being considered in the outer loop
				if = {
					limit = {
						meta_trigger = {
							text = {
								check_variable = { [LOGI_CAT]_logistics_motorisation_prio = priority }
							}
							LOGI_CAT = "[get_logistics_category]"
						}
					}
					# find how many trucks are needed in this category
					meta_effect = {
						text = {
							set_temp_variable = { truck_diff = total_[LOGI_CAT]_trucks }
							subtract_from_temp_variable = { truck_diff = [LOGI_CAT]_logi_trucks }
							if = {
								limit = { check_variable = { truck_diff > 0 } }
								# store in array with other categories of the same prioritisation
								add_to_temp_array = { curr_prio_trucks_needed_array = truck_diff }
								# store what category it is in this array, at the same index as the previous array
								add_to_temp_array = { curr_prio_trucks_needed_cat_array = logi_category }
							}
						}
						LOGI_CAT = "[get_logistics_category]"
					}
				}
			}
			# loop through each of the motorisation categories in the current priority, in order of increasing trucks needed
			while_loop_effect = {
				limit = {
					check_variable = { curr_prio_trucks_needed_array^num > 0 }
				}
				find_lowest_in_array = {
					array = curr_prio_trucks_needed_array
					value = curr_trucks_needed
					index = idx
				}
				# split the remaining number of trucks in the stockpile into equal parts for all categories of this prioritisation
				set_temp_variable = { trucks_to_transfer = trucks_remaining }
				divide_temp_variable = { trucks_to_transfer = curr_prio_trucks_needed_array^num }
				subtract_from_temp_variable = { trucks_to_transfer = 0.5 }
				round_temp_variable = trucks_to_transfer

				# if this category doesnt need all the trucks, we distribute them evenly to the remaining categories
				clamp_temp_variable = {
					var = trucks_to_transfer
					max = curr_trucks_needed
				}
				subtract_from_temp_variable = { trucks_remaining = trucks_to_transfer }
				
				# need this variable for the scripted localisation in the next effect
				meta_effect = {
					text = {
						set_temp_variable = { logi_category = curr_prio_trucks_needed_cat_array^[ARRAY_IDX] }
					}
					ARRAY_IDX = "[?idx]"
				}
				# update the logi truck stockpile for this category
				meta_effect = {
					text = {
						add_to_variable = { [LOGI_CAT]_logi_trucks = trucks_to_transfer }
					}
					LOGI_CAT = "[get_logistics_category]"
				}

				remove_from_temp_array = {
					array = curr_prio_trucks_needed_array
					index = idx
				}
				remove_from_temp_array = {
					array = curr_prio_trucks_needed_cat_array
					index = idx
				}
				if = {
					limit = { 
						check_variable = { trucks_remaining = 0 }
					}
					set_temp_variable = { break_loop = 1 }
				}
			}
		}
	}
	# update the actual truck stockpile
	subtract_from_temp_variable = { trucks_transferred = trucks_remaining }
	multiply_temp_variable = { trucks_transferred = -1 }
	add_equipment_to_stockpile = {
		type = motorized_equipment
		amount = trucks_transferred
	}

	calc_logi_truck_deficits = yes
}

excess_trucks_to_stockpile = {
	for_loop_effect = {
		end = num_logi_categories
		value = logi_category

		meta_effect = {
			text = {
				set_temp_variable = { truck_surplus = [LOGI_CAT]_logi_trucks }
				subtract_from_temp_variable = { truck_surplus = total_[LOGI_CAT]_trucks }
				if = {
					limit = {
						check_variable = { truck_surplus > 0 }
					}
					add_equipment_to_stockpile = {
						type = motorized_equipment
						amount = truck_surplus
					}
					subtract_from_variable = { [LOGI_CAT]_logi_trucks = truck_surplus }
				}
			}
			LOGI_CAT = "[get_logistics_category]"
		}
	}
}

calc_logi_truck_deficits = {
	set_variable = { total_logi_truck_deficit = 0 }
	set_variable = { total_truck_attrition_30_days = 0 }
	for_loop_effect = {
		end = num_logi_categories
		value = logi_category

		meta_effect = {
			text = {
				set_variable = { [LOGI_CAT]_logi_truck_deficit = [LOGI_CAT]_logi_trucks }
				subtract_from_variable = { [LOGI_CAT]_logi_truck_deficit = total_[LOGI_CAT]_trucks }
				add_to_variable = { total_logi_truck_deficit = [LOGI_CAT]_logi_truck_deficit }
				add_to_variable = { total_truck_attrition_30_days = [LOGI_CAT]_truck_attrition_30_days }
			}
			LOGI_CAT = "[get_logistics_category]"
		}
	}
	set_variable = { total_logi_trucks = total_trucks }
	add_to_variable = { total_logi_trucks = total_logi_truck_deficit }
}
