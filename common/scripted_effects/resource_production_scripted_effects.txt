# Per 1t Steel we need 1,1t Iron Ore, 1t Coal and 1kg Manganese + Silicon
@steel_refinery_cost_iron = 44
@steel_refinery_cost_coal = 8 
@steel_refinery_cost_chromium = 2	

# Per 1t Aluminium we need 5t Bauxite and 6,5t of Coal.
@aluminium_refinery_cost_bauxite = 64
@aluminium_refinery_cost_coal = 1

# Per 1t Fuel we need 12t Coal. Keep in mind that 1t Fuel ~ 1.200L Fuel
@synthetic_refinery_cost_coal = 24

# Per 1t Rubber we need 31,25t Coal
@rubber_refinery_cost_coal = 2

rp_close_steel_mills = {
    # How many need to be closed. Cannot simply use resource@, need to add resource_produced and resource_imported due to how engine and our refineries/mills work
    set_temp_variable = { num_to_swap_iron = resource_produced@iron }
    set_temp_variable = { num_to_swap_coal = resource_produced@coal }
    set_temp_variable = { num_to_swap_chromium = resource_produced@chromium }
    add_to_temp_variable = { num_to_swap_iron = resource_imported@iron }
    add_to_temp_variable = { num_to_swap_coal = resource_imported@coal }
    add_to_temp_variable = { num_to_swap_chromium = resource_imported@chromium }

    divide_temp_variable = { num_to_swap_iron = @steel_refinery_cost_iron }
    divide_temp_variable = { num_to_swap_coal = @steel_refinery_cost_coal }
    divide_temp_variable = { num_to_swap_chromium = @steel_refinery_cost_chromium }
    multiply_temp_variable = { num_to_swap_iron = -1 }
    multiply_temp_variable = { num_to_swap_coal = -1 }
    multiply_temp_variable = { num_to_swap_chromium = -1 }

    # round up
	add_to_temp_variable = { num_to_swap_iron = 0.5 }
	add_to_temp_variable = { num_to_swap_coal = 0.5 }
	add_to_temp_variable = { num_to_swap_chromium = 0.5 }
	round_temp_variable = num_to_swap_iron
	round_temp_variable = num_to_swap_coal
	round_temp_variable = num_to_swap_chromium

    set_temp_variable = { num_to_swap = 0 }
    
    if = {
        limit = {
            check_variable = { var = num_to_swap_iron value = num_to_swap_coal compare = greater_than_or_equals }
            check_variable = { var = num_to_swap_iron value = num_to_swap_chromium compare = greater_than_or_equals }
        }
        set_temp_variable = { num_to_swap = num_to_swap_iron }
    }
    if = {
        limit = {
            check_variable = { var = num_to_swap_coal value = num_to_swap_iron compare = greater_than_or_equals }
            check_variable = { var = num_to_swap_coal value = num_to_swap_chromium compare = greater_than_or_equals }
        }
        set_temp_variable = { num_to_swap = num_to_swap_coal }
    }
    if = {
        limit = {
            check_variable = { var = num_to_swap_chromium value = num_to_swap_iron compare = greater_than_or_equals }
            check_variable = { var = num_to_swap_chromium value = num_to_swap_coal compare = greater_than_or_equals }
        }
        set_temp_variable = { num_to_swap = num_to_swap_chromium }
    }
    set_temp_variable = { building_to_add = token:steel_refinery_inactive }
    set_temp_variable = { building_to_remove = token:steel_refinery }
    close_buildings = yes
}

rp_close_steel_mills_surplus = {
    # How many need to be closed to stop uneccessary raw material imports when in a surplus
    set_temp_variable = { num_to_swap_iron = resource_imported@iron }
    set_temp_variable = { num_to_swap_coal = resource_imported@coal }
    set_temp_variable = { num_to_swap_tungsten = resource_imported@tungsten }
    set_temp_variable = { num_to_swap_steel = resource@steel }
    subtract_from_temp_variable = { num_to_swap_iron = resource@iron }
    subtract_from_temp_variable = { num_to_swap_coal = resource@coal }
    subtract_from_temp_variable = { num_to_swap_tungsten = resource@tungsten }
    subtract_from_temp_variable = { num_to_swap_steel = resource_imported@steel }

    divide_temp_variable = { num_to_swap_iron = @steel_refinery_cost_iron }
    divide_temp_variable = { num_to_swap_coal = @steel_refinery_cost_coal }
    divide_temp_variable = { num_to_swap_tungsten = @steel_refinery_cost_tungsten }
    if = {
        limit = {
            original_tag = SOV
            NOT = { has_completed_focus = SOV_steel_casting_industry }
        }
        divide_temp_variable = { num_to_swap_steel = @steel_refinery_output_sov }
    }
    else = {
        divide_temp_variable = { num_to_swap_steel = @steel_refinery_output }
    }

    set_temp_variable = { temp_resource_gain = modifier@local_resources_factor }
    add_to_temp_variable = { temp_resource_gain = 1 }
    divide_temp_variable = { num_to_swap_steel = temp_resource_gain }

    # round up
	add_to_temp_variable = { num_to_swap_iron = 0.5 }
	add_to_temp_variable = { num_to_swap_coal = 0.5 }
	add_to_temp_variable = { num_to_swap_tungsten = 0.5 }
    add_to_temp_variable = { num_to_swap_steel = -0.5 } #round this one down
	round_temp_variable = num_to_swap_iron
	round_temp_variable = num_to_swap_coal
	round_temp_variable = num_to_swap_tungsten
    round_temp_variable = num_to_swap_steel

    set_temp_variable = { num_to_swap = num_to_swap_iron }
    if = {
        limit = { check_variable = { num_to_swap_coal > num_to_swap } }
        set_temp_variable = { num_to_swap = num_to_swap_coal }
    }
    if = {
        limit = { check_variable = { num_to_swap_tungsten > num_to_swap } }
        set_temp_variable = { num_to_swap = num_to_swap_tungsten }
    }
    if = {
        limit = { check_variable = { num_to_swap_steel < num_to_swap } }
        set_temp_variable = { num_to_swap = num_to_swap_steel }
    }

    if = {
        limit = { check_variable = { num_to_swap > 0 } }
        set_temp_variable = { building_to_add = token:steel_refinery_inactive }
        set_temp_variable = { building_to_remove = token:steel_refinery }
        close_buildings = yes
    }

    
}

# ARGUMENT VARIABLES, need to be set before
# token:building_to_add
# token:building_to_remove
open_buildings = {
    hidden_effect = {
        every_controlled_state = {# only add cores
            limit = {
                meta_trigger = {
                    text = {
                        [BUILDING] > 0
                    }
                    BUILDING = "[?building_to_remove.GetTokenKey]"
                }
                is_core_of = PREV
            }
            meta_effect = {
                text = {
                    add_to_temp_array = { [BUILDING]_state_candidates = THIS }
                    set_temp_variable = { local_buildings_to_swap = building_level@[BUILDING] }
                }
                BUILDING = "[?building_to_remove.GetTokenKey]"
            }
        }
    }
    swap_buildings = yes
}
# ARGUMENT VARIABLES, need to be set before
# token:building_to_add
# token:building_to_remove
close_buildings = {
    hidden_effect = {
        every_controlled_state = {# first add noncores
            limit = {
                meta_trigger = {
                    text = {
                        [BUILDING] > 0
                    }
                    BUILDING = "[?building_to_remove.GetTokenKey]"
                }
                NOT = { is_core_of = PREV }
            }
            meta_effect = {
                text = {
                    add_to_temp_array = { [BUILDING]_state_candidates = THIS }
                    set_temp_variable = { local_buildings_to_swap = building_level@[BUILDING] }
                }
                BUILDING = "[?building_to_remove.GetTokenKey]"
            }
        }
        every_controlled_state = {
            limit = {
                meta_trigger = {
                    text = {
                        [BUILDING] > 0
                    }
                    BUILDING = "[?building_to_remove.GetTokenKey]"
                }
                is_core_of = PREV
            }
            meta_effect = {
                text = {
                    add_to_temp_array = { [BUILDING]_state_candidates = THIS }
                    set_temp_variable = { local_buildings_to_swap = building_level@[BUILDING] }
                }
                BUILDING = "[?building_to_remove.GetTokenKey]"
            }
        }
    }
    swap_buildings = yes
}

# ARGUMENT VARIABLES, need to be set before
# token:building_to_add
# token:building_to_remove
swap_buildings = {
    log_building_swapping_start = yes
    meta_effect = {
        text = {
            for_each_loop = {
                array = [BUILDING_TO_REMOVE]_state_candidates
                break = break_var
                var:v = {
                    log_clamping = yes
                    clamp_temp_variable = {
                        var = local_buildings_to_swap
                        min = 0
                        max = num_to_swap
                    }
                    log_building_swapping_iteration_start = yes
                    remove_building = {
                        type = [BUILDING_TO_REMOVE]
                        level = local_buildings_to_swap
                    }
                    add_building_construction = {
                        type = [BUILDING_TO_ADD]
                        level = local_buildings_to_swap
                        instant_build = yes    
                    }
                    subtract_from_temp_variable = { num_to_swap = local_buildings_to_swap }
                    log_building_swapping_iteration_end = yes
                    if = {
                        limit = { 
                            check_variable = { var = num_to_swap value = 0 compare = less_than_or_equals }
                        }
                        set_temp_variable = { break_var = 1 }
                    }
                }
            }
            log_building_swapping_4 = yes
        }
        BUILDING_TO_REMOVE = "[?building_to_remove.GetTokenKey]"
        BUILDING_TO_ADD = "[?building_to_add.GetTokenKey]"
    }
}
